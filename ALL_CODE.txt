package com.astralis;

import com.astralis.command.AstralisCommand;
import com.astralis.core.*;

import com.astralis.listener.AbilityListener;

import net.kyori.adventure.text.Component;
import net.kyori.adventure.text.minimessage.MiniMessage;
import org.bukkit.Bukkit;
import org.bukkit.World;
import org.bukkit.command.PluginCommand;
import org.bukkit.entity.Player;
import org.bukkit.plugin.java.JavaPlugin;

/**
 * Astralis - constellation sigils with Stardust energy, a rock-paper-scissors counter ring,
 * and cinematic abilities.
 */
public final class Astralis extends JavaPlugin {
    public static final MiniMessage MM = MiniMessage.miniMessage();
    private static Astralis instance;

    private Keys keys;
    private SigilManager sigils;
    private StardustManager stardust;
    private CooldownManager cooldowns;
    private StatusManager statuses;
    private TrustManager trust;
    private AbilityManager abilities;
    private EventManager events;

    public static Astralis get() { return instance; }
    public static Component mm(String s) { return MM.deserialize(s); }

    @Override
    public void onEnable() {
        instance = this;
        saveDefaultConfig();

        keys = new Keys(this);
        sigils = new SigilManager(this);
        stardust = new StardustManager(this);
        cooldowns = new CooldownManager();
        statuses = new StatusManager();
        trust = new TrustManager(this);
        abilities = new AbilityManager(this);
        events = new EventManager(this);

        sigils.registerRecipes();

        getServer().getPluginManager().registerEvents(new AbilityListener(this), this);
        getServer().getPluginManager().registerEvents(new CombatListener(this), this);
        getServer().getPluginManager().registerEvents(new SigilListener(this), this);

        PluginCommand cmd = getCommand("astralis");
        if (cmd != null) {
            AstralisCommand ac = new AstralisCommand(this);
            cmd.setExecutor(ac);
            cmd.setTabCompleter(ac);
        }

        new StarTask(this).runTaskTimer(this, 20L, 2L);
        events.startScheduler();

        for (Player p : Bukkit.getOnlinePlayers()) {
            stardust.load(p);
            trust.load(p);
            sigils.discover(p);
        }
        getLogger().info("Astralis awakened. The stars are aligned.");
    }

    @Override
    public void onDisable() {
        if (events != null) events.stop(null);
        for (Player p : Bukkit.getOnlinePlayers()) {
            stardust.save(p);
            trust.save(p);
        }
        if (sigils != null) sigils.unregisterRecipes();
    }

    public void reload() {
        reloadConfig();
        sigils.unregisterRecipes();
        sigils.registerRecipes();
    }

    // ---- config helpers -------------------------------------------------------------
    /** Reads a number under "constellations." */
    public double val(String path, double def) { return getConfig().getDouble("constellations." + path, def); }

    public double tierPotency(Tier t) {
        return getConfig().getDouble("tiers." + t.name().toLowerCase() + ".potency", t == Tier.RADIANT ? 1.3 : 1.0);
    }

    public double tierCooldownMult(Tier t) {
        return getConfig().getDouble("tiers." + t.name().toLowerCase() + ".cooldown-multiplier", t == Tier.RADIANT ? 0.8 : 1.0);
    }

    public int ultCost(Tier t) {
        return getConfig().getInt("tiers." + t.name().toLowerCase() + ".ultimate-cost", t == Tier.RADIANT ? 85 : 100);
    }

    public boolean disabledWorld(World w) { return getConfig().getStringList("general.disabled-worlds").contains(w.getName()); }

    public boolean enabled(Constellation c) {
        return getConfig().getBoolean("constellations." + c.name().toLowerCase() + ".enabled", true);
    }

    // ---- accessors ------------------------------------------------------------------
    public Keys keys() { return keys; }
    public SigilManager sigils() { return sigils; }
    public StardustManager stardust() { return stardust; }
    public CooldownManager cooldowns() { return cooldowns; }
    public StatusManager statuses() { return statuses; }
    public TrustManager trust() { return trust; }
    public AbilityManager abilities() { return abilities; }
    public EventManager events() { return events; }
}

        Constellation c = Constellation.parse(name);
        if (c == null) msg(s, "<red>Unknown constellation: <white>" + esc(name) + "<gray>. Try: <white>" + names());
        return c;
    }

    private static String names() {
        StringBuilder sb = new StringBuilder();
        for (Constellation c : Constellation.values()) { if (sb.length() > 0) sb.append(", "); sb.append(c.display().toLowerCase()); }
        return sb.toString();
    }

    // ------------------------------------------------------------------ dispatch
    @Override
    public boolean onCommand(CommandSender s, Command cmd, String label, String[] a) {
        if (a.length == 0) { help(s); return true; }
        switch (a[0].toLowerCase(Locale.ROOT)) {
            case "help", "?" -> help(s);
            case "info", "list" -> info(s, a);
            case "stats" -> stats(s, a);
            case "trust" -> trust(s, a, true);
            case "untrust" -> trust(s, a, false);
            case "trusted" -> trusted(s);
            case "event", "events" -> event(s, a);
            case "give" -> { if (admin(s)) give(s, a); }
            case "core" -> { if (admin(s)) core(s, a); }
            case "tier" -> { if (admin(s)) tier(s, a); }
            case "remove" -> { if (admin(s)) remove(s, a); }
            case "awaken" -> { if (admin(s)) awaken(s, a); }
            case "reroll" -> { if (admin(s)) reroll(s, a); }
            case "energy" -> { if (admin(s)) energy(s, a); }
            case "cooldown", "cd" -> { if (admin(s)) cooldown(s, a); }
            case "cast" -> { if (admin(s)) cast(s, a); }
            case "holders" -> { if (admin(s)) holders(s); }
            case "reload" -> { if (admin(s)) { plugin.reload(); msg(s, "<green>Configuration reloaded."); } }
            case "selftest" -> { if (admin(s)) selftest(s); }
            default -> msg(s, "<red>Unknown subcommand. <gray>Try <white>/astralis help");
        }
        return true;
    }

    private void stats(CommandSender s, String[] a) {
        Player t;
        if (a.length > 1) { t = online(s, a[1]); if (t == null) return; }
        else if (s instanceof Player p) t = p;
        else { msg(s, "<red>Specify a player."); return; }

        SigilManager sm = plugin.sigils();
        Sigil sg = sm.active(t);
        raw(s, "<gradient:#5ad7ff:#b57bff><bold>✦ " + esc(t.getName()) + "</bold></gradient>");
        if (sg == null) { raw(s, " <gray>No sigil."); return; }
        Constellation c = sg.constellation();
        raw(s, " " + c.tag() + "✦ " + c.display() + " <gray>(" + sg.tier().label() + ")  potency <white>"
                + String.format("%.2f", sm.potency(t, sg)) + (sg.owner() != null && !sg.owner().equals(t.getUniqueId()) ? " <red>(unattuned)" : ""));
        raw(s, " <gray>Stardust: <white>" + (int) plugin.stardust().get(t) + "<gray>/" + (int) plugin.stardust().max()
                + " <dark_gray>(Zenith costs " + plugin.ultCost(sg.tier()) + ")");
        UUID id = t.getUniqueId();
        raw(s, " <gray>Primary: <white>" + fmt(plugin.cooldowns().remaining(id, c.name() + ":P"))
                + " <gray>| Zenith: <white>" + fmt(plugin.cooldowns().remaining(id, c.name() + ":U")));
        raw(s, " <gray>Carrying <white>" + sm.carried(t) + "<gray>/" + plugin.getConfig().getInt("general.max-sigils-carried", 2) + " sigils");
    }

    private static String fmt(double sec) { return sec <= 0 ? "ready" : String.format("%.1fs", sec); }

    // ------------------------------------------------------------------ trust
    private void trust(CommandSender s, String[] a, boolean add) {
        if (!(s instanceof Player p)) { msg(s, "<red>Players only."); return; }
        if (a.length < 2) { msg(s, "<gray>Usage: <white>/astralis " + (add ? "trust" : "untrust") + " [player]"); return; }
        OfflinePlayer t = Bukkit.getPlayerExact(a[1]);
        if (t == null) t = Bukkit.getOfflinePlayerIfCached(a[1]);
        if (t == null) { msg(s, "<red>I don't know a player called <white>" + esc(a[1])); return; }

                }

            case "locate" -> {
                if (s instanceof Player p) msg(s, ev.locate(p));
                else msg(s, "<red>Players only.");
            }
            case "start" -> {
                if (!admin(s)) return;

                EventManager.Type t = EventManager.Type.parse(a[2]);

            }

        }

    // ------------------------------------------------------------------ selftest
    private void selftest(CommandSender s) {
        int pass = 0, fail = 0;
        List<String> report = new ArrayList<>();
        SigilManager sm = plugin.sigils();

        // 1. every sigil round-trips through item NBT
        for (Constellation c : Constellation.values()) {
            for (Tier t : Tier.values()) {
                UUID owner = UUID.randomUUID();
                ItemStack it = sm.create(c, t, owner, "Tester");
                Sigil r = sm.read(it);
                boolean ok = r != null && r.constellation() == c && r.tier() == t && owner.equals(r.owner()) && r.id() != null;
                if (ok) pass++; else { fail++; report.add("round-trip failed: " + c + "/" + t); }
            }
        }
        // 2. counter ring is a clean 5-cycle; mythicals are outside it
        for (Constellation c : Constellation.ring()) {
            boolean ok = c.prey().predator() == c && c.predator().prey() == c && c.prey() != c;
            if (ok) pass++; else { fail++; report.add("ring broken at " + c); }
        }
        Constellation walk = Constellation.ring().get(0);
        for (int i = 0; i < Constellation.ring().size(); i++) walk = walk.prey();
        if (walk == Constellation.ring().get(0)) pass++; else { fail++; report.add("ring does not close after 5 steps"); }
        for (Constellation c : Constellation.mythics()) {
            if (c.prey() == null && c.predator() == null) pass++; else { fail++; report.add(c + " should be outside the ring"); }
        }
        // 3. the Stellar Core is recognised and is not mistaken for a sigil
        ItemStack core = sm.createCore();
        if (sm.isCore(core) && !sm.isSigil(core)) pass++; else { fail++; report.add("core item detection"); }
        // 4. cooldowns & statuses
        UUID id = UUID.randomUUID();
        plugin.cooldowns().set(id, "T", 5);
        double rem = plugin.cooldowns().remaining(id, "T");
        if (rem > 4.5 && rem <= 5.0 && !plugin.cooldowns().ready(id, "T")) pass++; else { fail++; report.add("cooldown timing"); }
        plugin.cooldowns().clear(id);
        plugin.statuses().apply(id, StatusManager.Status.ECHO, 40, null);
        if (plugin.statuses().has(id, StatusManager.Status.ECHO)) pass++; else { fail++; report.add("status apply"); }
        plugin.statuses().clearAll(id);
        if (!plugin.statuses().has(id, StatusManager.Status.ECHO)) pass++; else { fail++; report.add("status clear"); }
        // 5. config completeness
        for (String path : CONFIG_PATHS) {
            if (plugin.getConfig().contains(path)) pass++; else { fail++; report.add("missing config key: " + path); }
        }
        // 6. every constellation has a kit
        for (Constellation c : Constellation.values()) {
            if (plugin.abilities().kit(c) != null) pass++; else { fail++; report.add("no kit for " + c); }
        }

        msg(s, (fail == 0 ? "<green>" : "<red>") + "Selftest: " + pass + " passed, " + fail + " failed.");
        for (String r : report) raw(s, " <red>✘ <gray>" + esc(r));
    }

    private static int parseInt(String s, int def) { try { return Integer.parseInt(s); } catch (NumberFormatException e) { return def; } }
    private static double parseDouble(String s, double def) { try { return Double.parseDouble(s); } catch (NumberFormatException e) { return def; } }

    // ------------------------------------------------------------------ tab completion
    @Override
    public List<String> onTabComplete(CommandSender s, Command cmd, String label, String[] a) {
        boolean adm = s.hasPermission("astralis.admin");
        List<String> out = new ArrayList<>();
        if (a.length == 1) {
            out.addAll(PLAYER_SUBS);
            if (adm) out.addAll(ADMIN_SUBS);
        } else {
            String sub = a[0].toLowerCase(Locale.ROOT);
            switch (a.length) {
                case 2 -> {
                    switch (sub) {
                        case "info" -> addConstellations(out);
                        case "trust", "untrust", "stats" -> addPlayers(out);
                        case "event", "events" -> { out.addAll(EVENT_SUBS); }
                        default -> { if (adm && ADMIN_SUBS.contains(sub) && !sub.equals("reload") && !sub.equals("holders") && !sub.equals("selftest")) addPlayers(out); }
                    }
                }
                case 3 -> {
                    if (!adm) break;
                    switch (sub) {
                        case "give", "remove", "awaken" -> addConstellations(out);
                        case "tier" -> out.addAll(List.of("dim", "radiant"));
                        case "energy" -> out.addAll(List.of("set", "add", "take", "max", "reset"));
                        case "cast" -> out.addAll(List.of("primary", "ultimate"));
                        case "event", "events" -> { if (a[1].equalsIgnoreCase("start")) for (EventManager.Type t : EventManager.Type.values()) out.add(t.name().toLowerCase()); }
                        default -> { }
                    }
                }
                case 4 -> {
                    if (adm && sub.equals("give")) out.addAll(List.of("dim", "radiant"));
                    if (adm && (sub.equals("event") || sub.equals("events")) && a[1].equalsIgnoreCase("start")) out.addAll(List.of("5", "10", "20"));
                }
                default -> { }
            }
        }
        String last = a[a.length - 1].toLowerCase(Locale.ROOT);
        out.removeIf(x -> !x.toLowerCase(Locale.ROOT).startsWith(last));
        return out;
    }

    private void addConstellations(List<String> out) { for (Constellation c : Constellation.values()) out.add(c.display().toLowerCase()); }
    private void addPlayers(List<String> out) { for (Player p : Bukkit.getOnlinePlayers()) out.add(p.getName()); }
}

package com.astralis.core;

import com.astralis.Astralis;
import com.astralis.kit.*;
import net.kyori.adventure.text.Component;
import net.kyori.adventure.text.format.NamedTextColor;
import org.bukkit.GameMode;
import org.bukkit.Sound;
import org.bukkit.entity.Player;

import java.util.EnumMap;
import java.util.Map;
import java.util.UUID;

/** Gatekeeper for casting: cooldowns, Stardust cost, world rules and feedback. */
public final class AbilityManager {
    private final Astralis plugin;
    private final Map<Constellation, Kit> kits = new EnumMap<>(Constellation.class);

    public AbilityManager(Astralis plugin) {
        this.plugin = plugin;
        kits.put(Constellation.ARIES, new AriesKit(plugin));
        kits.put(Constellation.ORION, new OrionKit(plugin));
        kits.put(Constellation.DRACO, new DracoKit(plugin));
        kits.put(Constellation.CASSIOPEIA, new CassiopeiaKit(plugin));
        kits.put(Constellation.GEMINI, new GeminiKit(plugin));
        kits.put(Constellation.PHOENIX, new PhoenixKit(plugin));
        kits.put(Constellation.ANDROMEDA, new AndromedaKit(plugin));
    }

    public Kit kit(Constellation c) { return kits.get(c); }

    private boolean canCast(Player p, Sigil s) {
        if (p.isDead() || p.getGameMode() == GameMode.SPECTATOR) return false;
        if (plugin.disabledWorld(p.getWorld())) { deny(p, "Sigils are dormant in this world."); return false; }
        if (!plugin.enabled(s.constellation())) { deny(p, "This constellation is disabled."); return false; }
        return true;
    }

    public void primary(Player p) {
        Sigil s = plugin.sigils().read(p.getInventory().getItemInMainHand());
        if (s == null || !canCast(p, s)) return;
        UUID id = p.getUniqueId();
        String key = s.constellation().name() + ":P";
        double left = plugin.cooldowns().remaining(id, key);
        if (left > 0) { deny(p, String.format("%s ready in %.1fs", s.constellation().primaryName(), left)); return; }

        Kit kit = kits.get(s.constellation());
        if (kit.primary(p, plugin.sigils().potency(p, s))) {
            double cd = kit.primaryBaseCooldown() * plugin.tierCooldownMult(s.tier());
            plugin.cooldowns().set(id, key, cd);
            p.setCooldown(s.constellation().material(), (int) (cd * 20));
        }
    }

    public void ultimate(Player p) {
        Sigil s = plugin.sigils().read(p.getInventory().getItemInMainHand());
        if (s == null || !canCast(p, s)) return;
        UUID id = p.getUniqueId();
        Constellation c = s.constellation();
        String key = c.name() + ":U";

        double left = plugin.cooldowns().remaining(id, key);
        if (left > 0) { deny(p, String.format("Zenith ready in %.0fs", Math.ceil(left))); return; }
        int cost = plugin.ultCost(s.tier());
        double have = plugin.stardust().get(p);
        if (have < cost) { deny(p, String.format("Not enough Stardust (%d/%d). Fight to gather more.", (int) have, cost)); return; }

        Kit kit = kits.get(c);
        if (kit.ultimate(p, plugin.sigils().potency(p, s))) {
            plugin.stardust().add(p, -cost);
            plugin.cooldowns().set(id, key, plugin.getConfig().getDouble("general.ultimate-cooldown", 30));
            p.sendActionBar(c.text("★ ZENITH — " + c.ultName() + " ★"));
            p.getWorld().playSound(p.getLocation(), Sound.BLOCK_BEACON_ACTIVATE, 1.2f, 1.7f);
        }
    }

    private void deny(Player p, String msg) {
        p.sendActionBar(Component.text(msg, NamedTextColor.RED));
        p.playSound(p.getLocation(), Sound.ENTITY_VILLAGER_NO, 0.6f, 1.2f);
    }
}

public final class CooldownManager {
    private final Map<UUID, Map<String, Long>> map = new HashMap<>();

    public boolean ready(UUID id, String key) { return remaining(id, key) <= 0; }

    /** Remaining cooldown in seconds. */
    public double remaining(UUID id, String key) {
        Map<String, Long> m = map.get(id);
        if (m == null) return 0;
        Long until = m.get(key);
        if (until == null) return 0;
        return Math.max(0, (until - System.currentTimeMillis()) / 1000.0);
    }

    public void set(UUID id, String key, double seconds) {
        map.computeIfAbsent(id, k -> new HashMap<>()).put(key, System.currentTimeMillis() + (long) (seconds * 1000));
    }

    public void clear(UUID id) { map.remove(id); }
}

package com.astralis.core;

import com.astralis.Astralis;
import org.bukkit.GameMode;
import org.bukkit.entity.*;

/** All ability damage flows through here so the counter ring and marks apply uniformly. */
public final class Damage {
    private static boolean abilityDamage;

    private Damage() { }

    /** True while an ability is dealing damage (listeners use this to avoid recursion). */
    public static boolean isAbility() { return abilityDamage; }

    public static boolean valid(Player caster, Entity e) {
        if (e == caster || !(e instanceof LivingEntity le)) return false;
        if (le.isDead() || !le.isValid() || le instanceof ArmorStand) return false;
        if (le instanceof Tameable t && t.isTamed()) return false;
        if (le instanceof Player other) {
            if (other.getGameMode() == GameMode.SPECTATOR) return false;
            if (Astralis.get().trust().trusts(caster.getUniqueId(), other.getUniqueId())) return false;
        }
        return true;
    }

    /** Counter-ring multiplier x Orion mark multiplier. */
    public static double modifier(Player attacker, LivingEntity victim) {
        Astralis pl = Astralis.get();
        double m = pl.sigils().counterMultiplier(attacker, victim);
        StatusManager st = pl.statuses();
        if (st.has(victim.getUniqueId(), StatusManager.Status.MARKED)
                && attacker.getUniqueId().equals(st.source(victim.getUniqueId(), StatusManager.Status.MARKED))) {
            m *= 1 + pl.val("orion.beam.mark-bonus", 0.10);
        }
        return m;
    }

    public static boolean deal(Player caster, LivingEntity target, double amount) {
        if (amount <= 0 || !valid(caster, target)) return false;
        double dmg = amount * modifier(caster, target) * Astralis.get().getConfig().getDouble("general.ability-damage-scale", 1.0);
        boolean prev = abilityDamage;
        abilityDamage = true;
        try {
            target.setNoDamageTicks(0);
            target.damage(dmg, caster);
        } finally {
            abilityDamage = prev;
        }
        return true;
    }
}

        return pool.get(ThreadLocalRandom.current().nextInt(pool.size()));

package com.astralis.core;

import java.util.EnumMap;
import java.util.HashMap;
import java.util.Iterator;
import java.util.Map;
import java.util.UUID;

/** Lightweight timed statuses that abilities attach to entities. */
public final class StatusManager {
    public enum Status { MARKED, ECHO, AEGIS, NO_FALL }

    private record Entry(long expires, UUID source) { }

    private final Map<UUID, EnumMap<Status, Entry>> map = new HashMap<>();

    public void apply(UUID id, Status s, int ticks, UUID source) {
        map.computeIfAbsent(id, k -> new EnumMap<>(Status.class))
                .put(s, new Entry(System.currentTimeMillis() + ticks * 50L, source));
    }

    public boolean has(UUID id, Status s) {
        EnumMap<Status, Entry> m = map.get(id);
        if (m == null) return false;
        Entry e = m.get(s);
        if (e == null) return false;
        if (e.expires() < System.currentTimeMillis()) {
            m.remove(s);
            return false;
        }
        return true;
    }

    public UUID source(UUID id, Status s) {
        if (!has(id, s)) return null;
        return map.get(id).get(s).source();
    }

    public void clear(UUID id, Status s) {
        EnumMap<Status, Entry> m = map.get(id);
        if (m != null) m.remove(s);
    }

    public void clearAll(UUID id) { map.remove(id); }

    public void purge() {
        long now = System.currentTimeMillis();
        Iterator<Map.Entry<UUID, EnumMap<Status, Entry>>> it = map.entrySet().iterator();
        while (it.hasNext()) {
            EnumMap<Status, Entry> m = it.next().getValue();
            m.values().removeIf(e -> e.expires() < now);
            if (m.isEmpty()) it.remove();
        }
    }
}

package com.astralis.core;

import org.bukkit.FluidCollisionMode;
import org.bukkit.Location;
import org.bukkit.World;
import org.bukkit.entity.Entity;
import org.bukkit.entity.LivingEntity;
import org.bukkit.entity.Player;
import org.bukkit.util.BoundingBox;
import org.bukkit.util.RayTraceResult;
import org.bukkit.util.Vector;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

public final class Targeting {
    private Targeting() { }

    public record Pierce(List<LivingEntity> entities, Location end) { }

    /** Ray that stops at solid blocks but passes through every entity it touches. */
    public static Pierce pierce(Player p, double range, double inflate) {
        Location eye = p.getEyeLocation();
        World w = eye.getWorld();
        Vector dir = eye.getDirection().normalize();
        RayTraceResult br = w.rayTraceBlocks(eye, dir, range, FluidCollisionMode.NEVER, true);
        double max = br != null ? br.getHitPosition().distance(eye.toVector()) : range;
        Location end = eye.clone().add(dir.clone().multiply(max));
        Location mid = eye.clone().add(dir.clone().multiply(max / 2));
        double half = max / 2 + 2;

        List<LivingEntity> out = new ArrayList<>();
        for (Entity e : w.getNearbyEntities(mid, half, half, half)) {
            if (!Damage.valid(p, e)) continue;
            LivingEntity le = (LivingEntity) e;
            BoundingBox box = le.getBoundingBox().expand(inflate);
            if (box.rayTrace(eye.toVector(), dir, max) != null) out.add(le);
        }
        out.sort(Comparator.comparingDouble(le -> le.getLocation().distanceSquared(eye)));
        return new Pierce(out, end);
    }

    /** Where the player is aiming (entity or block), or null if aiming at the sky. */
    public static Location aim(Player p, double range) {
        Location eye = p.getEyeLocation();
        RayTraceResult r = p.getWorld().rayTrace(eye, eye.getDirection(), range, FluidCollisionMode.NEVER, true, 0.4,
                e -> Damage.valid(p, e));
        if (r == null) return null;
        if (r.getHitEntity() != null) return r.getHitEntity().getLocation();
        Vector v = r.getHitPosition();
        return new Location(p.getWorld(), v.getX(), v.getY(), v.getZ());
    }
}

/** One-way trust list: your abilities never hurt players you trust. */
public final class TrustManager {
    private final Astralis plugin;
    private final Map<UUID, Set<UUID>> map = new HashMap<>();

    public TrustManager(Astralis plugin) { this.plugin = plugin; }

    public void load(Player p) {
        Set<UUID> set = new LinkedHashSet<>();
        String raw = p.getPersistentDataContainer().get(plugin.keys().trusted, PersistentDataType.STRING);
        if (raw != null && !raw.isBlank()) {
            for (String s : raw.split(",")) {
                try { set.add(UUID.fromString(s.trim())); } catch (IllegalArgumentException ignored) { }
            }
        }
        map.put(p.getUniqueId(), set);
    }

    public void save(Player p) {
        Set<UUID> set = map.remove(p.getUniqueId());
        if (set == null) return;
        StringBuilder sb = new StringBuilder();
        for (UUID u : set) { if (sb.length() > 0) sb.append(','); sb.append(u); }
        p.getPersistentDataContainer().set(plugin.keys().trusted, PersistentDataType.STRING, sb.toString());
    }

    public boolean trusts(UUID owner, UUID other) {
        Set<UUID> s = map.get(owner);
        return s != null && s.contains(other);
    }

    public boolean add(Player owner, UUID other) {
        return map.computeIfAbsent(owner.getUniqueId(), k -> new LinkedHashSet<>()).add(other);
    }

    public boolean remove(Player owner, UUID other) {
        Set<UUID> s = map.get(owner.getUniqueId());
        return s != null && s.remove(other);
    }

    public Set<UUID> list(UUID owner) { return map.getOrDefault(owner, Collections.emptySet()); }
}

    private void autoTick() {
        var cfg = plugin.getConfig();
        if (!cfg.getBoolean("events.auto.enabled", false) || active != null) return;
        if (Bukkit.getOnlinePlayers().size() < cfg.getInt("events.auto.min-players", 2)) return;
        long now = System.currentTimeMillis();
        long gap = cfg.getInt("events.auto.interval-minutes", 90) * 60000L;
        if (nextAuto == 0) { nextAuto = now + gap; return; }
        if (now < nextAuto) return;
        Type[] ts = Type.values();
        start(ts[ThreadLocalRandom.current().nextInt(ts.length)], 0);
        nextAuto = now + gap;
    }

    private boolean pickZone() {

        if (w == null) return false;
        Location sp = w.getSpawnLocation();
        double min = cfgD(Type.CONVERGENCE, "min-distance", 150), max = Math.max(min + 1, cfgD(Type.CONVERGENCE, "max-distance", 600));

    private void tickConvergence() {
        if (zone == null) { stop(null); return; }
        World w = zone.getWorld();
        double r = cfgD(Type.CONVERGENCE, "zone-radius", 6);
        int need = cfgI(Type.CONVERGENCE, "capture-seconds", 90) * 20;

        // visuals
        Fx.ring(zone.clone().add(0, 0.2, 0), r, 48, Particle.DUST, Fx.dust(Color.fromRGB(0xb57bff), 1.5f));
        for (double y = 0; y < 30; y += 1.5) w.spawnParticle(Particle.END_ROD, zone.clone().add(0, y, 0), 1, 0.15, 0, 0.15, 0);
        if (elapsed % 6000 == 0) {
            announce("<light_purple>✦ <gray>The Convergence burns at <white>" + zone.getBlockX() + ", " + zone.getBlockZ() + "<gray>.");
        }

        // who is standing in the zone?
        List<Player> in = new ArrayList<>();
        for (Player p : eligible()) {
            if (p.getWorld() != w) continue;
            double dx = p.getLocation().getX() - zone.getX(), dz = p.getLocation().getZ() - zone.getZ();
            if (dx * dx + dz * dz <= r * r && Math.abs(p.getLocation().getY() - zone.getY()) <= 10) in.add(p);
        }
        UUID holder = null;
        if (in.size() == 1) {
            Player c = in.get(0);
            holder = c.getUniqueId();
            int v = progress.merge(holder, 10, Integer::sum);
            if (v >= need) { win(c); return; }
        }
        // everyone who is not actively capturing slowly loses progress
        Iterator<Map.Entry<UUID, Integer>> it = progress.entrySet().iterator();
        while (it.hasNext()) {
            Map.Entry<UUID, Integer> en = it.next();
            if (en.getKey().equals(holder)) continue;
            en.setValue(Math.max(0, en.getValue() - 2));
            if (en.getValue() == 0) it.remove();
        }

        // bossbar
        UUID leader = null;
        int best = 0;
        for (Map.Entry<UUID, Integer> en : progress.entrySet()) if (en.getValue() > best) { best = en.getValue(); leader = en.getKey(); }
        String where = zone.getBlockX() + ", " + zone.getBlockZ();
        String title;
        if (in.size() > 1) title = "<red><bold>CONTESTED</bold> <dark_gray>— <gray>only one player may hold the zone";
        else if (leader == null) title = "<light_purple><bold>Convergence</bold> <dark_gray>@ <white>" + where + " <dark_gray>— <gray>unclaimed";
        else {
            OfflinePlayer op = Bukkit.getOfflinePlayer(leader);
            String name = op.getName() == null ? "?" : Astralis.MM.escapeTags(op.getName());
            title = "<light_purple><bold>Convergence</bold> <dark_gray>— <white>" + name + " <gold>" + (int) (100.0 * best / need) + "%";
        }
        bar.name(Astralis.mm(title));
        bar.progress((float) Math.min(1.0, best / (double) need));
    }

    private void win(Player winner) {
        List<Constellation> pool = new ArrayList<>();
        for (Constellation c : Constellation.mythics()) if (plugin.enabled(c)) pool.add(c);
        if (pool.isEmpty()) { stop("The Convergence fades. (No Mythicals are enabled.)"); return; }
        Constellation c = pool.get(ThreadLocalRandom.current().nextInt(pool.size()));
        plugin.sigils().giveOrDrop(winner, plugin.sigils().create(c, Tier.DIM, winner.getUniqueId(), winner.getName()));

        Location l = winner.getLocation().add(0, 1, 0);
        winner.getWorld().spawnParticle(Particle.TOTEM_OF_UNDYING, l, 120, 0.6, 1.2, 0.6, 0.6);
        Fx.sound(winner.getLocation(), Sound.UI_TOAST_CHALLENGE_COMPLETE, 2f, 1f);
        stop(null);
        Bukkit.broadcast(Astralis.mm("<gold>✦ <white>" + Astralis.MM.escapeTags(winner.getName())
                + " <gray>has claimed the Mythical " + c.tag() + "<bold>" + c.display() + "</bold><gray>, " + c.title() + "!"));
        for (Player p : Bukkit.getOnlinePlayers()) p.playSound(p.getLocation(), Sound.ENTITY_ENDER_DRAGON_GROWL, 0.6f, 1.4f);
    }

    // ------------------------------------------------------------------ ECLIPSE
    private void tickEclipse() {
        if (elapsed % 40 != 0) return;
        for (Player p : eligible()) {
            if (plugin.sigils().active(p) == null) continue;
            p.getWorld().spawnParticle(Particle.END_ROD, p.getLocation().add(0, 1, 0), 8, 0.4, 0.8, 0.4, 0.06);
        }
    }

    // ------------------------------------------------------------------ info
    private static String compass(int dx, int dz) {
        double ang = Math.atan2(dz, dx);
        int idx = (int) Math.round(ang / (Math.PI / 4));
        return COMPASS[((idx % 8) + 8) % 8];
    }

    public String status() {
        if (active == null) return "<gray>No event is running.";
        int left = Math.max(0, (total - elapsed) / 20);
        return "<white>" + active.label + " <gray>— " + String.format("%d:%02d", left / 60, left % 60) + " left. <dark_gray>" + active.blurb;
    }

    public String locate(Player p) {
        if (active != Type.CONVERGENCE || zone == null) return "<gray>There is no Convergence to locate right now.";
        if (p.getWorld() != zone.getWorld()) return "<gray>The Convergence is in another world: <white>" + zone.getWorld().getName();
        int dx = zone.getBlockX() - p.getLocation().getBlockX(), dz = zone.getBlockZ() - p.getLocation().getBlockZ();
        int dist = (int) Math.sqrt((double) dx * dx + (double) dz * dz);
        return "<light_purple>✦ <gray>The Convergence is <white>" + dist + " blocks <gray>to the <white>" + compass(dx, dz)
                + " <dark_gray>(" + zone.getBlockX() + ", " + zone.getBlockZ() + ")";
    }

    public static void play(Astralis plugin, Player p, Constellation c, Runnable done) {
        p.showTitle(Title.title(
                Component.text("✦ AWAKENING ✦", c.color(), TextDecoration.BOLD),

                Title.Times.times(Duration.ofMillis(400), Duration.ofMillis(3000), Duration.ofMillis(800))));

        Vector f = p.getEyeLocation().getDirection().setY(0);
        if (f.lengthSquared() < 0.01) f = new Vector(0, 0, 1);
        final Vector forward = f.normalize();
        final Vector right = forward.clone().crossProduct(new Vector(0, 1, 0)).normalize();
        final Location origin = p.getLocation().add(forward.clone().multiply(8)).add(0, 6, 0);
        final double[][] st = c.stars();
        final int[][] lk = c.links();
        final org.bukkit.Particle.DustOptions dust = Fx.dust(c.bukkitColor(), 1.5f);
        final int total = st.length * 8 + 50;

        Fx.sound(p.getLocation(), Sound.BLOCK_BEACON_POWER_SELECT, 1f, 0.6f);

        new BukkitRunnable() {
            int t = 0;

            Location at(int i) {
                return origin.clone().add(right.clone().multiply(st[i][0])).add(0, st[i][1], 0);
            }

            @Override
            public void run() {
                if (!p.isOnline()) { cancel(); return; }
                World w = p.getWorld();
                int revealed = Math.min(st.length, (t + 7) / 8);

                if (t % 8 == 1 && t / 8 < st.length) {
                    p.playSound(p.getLocation(), Sound.BLOCK_AMETHYST_BLOCK_CHIME, 1f, 0.7f + 0.12f * (t / 8));
                }
                for (int i = 0; i < revealed; i++) {
                    Location l = at(i);
                    w.spawnParticle(Particle.END_ROD, l, 2, 0.08, 0.08, 0.08, 0);
                    w.spawnParticle(Particle.DUST, l, 3, 0.15, 0.15, 0.15, 0, dust);
                }

package com.astralis.fx;

import org.bukkit.Color;
import org.bukkit.Location;
import org.bukkit.Particle;
import org.bukkit.Sound;
import org.bukkit.World;
import org.bukkit.util.Vector;

/** Particle & sound helpers shared by every ability. */
public final class Fx {
    private Fx() { }

    public static Particle.DustOptions dust(Color c, float size) { return new Particle.DustOptions(c, size); }

    /** Spawns a particle, passing data only when it is non-null. */
    public static void spawn(World w, Particle p, Location l, int count, double dx, double dy, double dz, double speed, Object data) {
        if (data == null) w.spawnParticle(p, l, count, dx, dy, dz, speed);
        else w.spawnParticle(p, l, count, dx, dy, dz, speed, data);
    }

    public static void ring(Location c, double r, int points, Particle p, Object data) {
        World w = c.getWorld();
        for (int i = 0; i < points; i++) {
            double a = 2 * Math.PI * i / points;
            spawn(w, p, c.clone().add(Math.cos(a) * r, 0, Math.sin(a) * r), 1, 0, 0, 0, 0, data);
        }
    }

    public static void line(Location a, Location b, double step, Particle p, Object data) {
        World w = a.getWorld();
        Vector d = b.toVector().subtract(a.toVector());
        double len = d.length();
        if (len < 0.01) return;
        d.normalize().multiply(step);
        Location cur = a.clone();
        for (double t = 0; t < len; t += step) {
            spawn(w, p, cur, 1, 0, 0, 0, 0, data);
            cur.add(d);
        }
    }

    /** Two unit vectors perpendicular to dir (and each other). */
    public static Vector[] basis(Vector dir) {
        Vector up = Math.abs(dir.getY()) > 0.95 ? new Vector(1, 0, 0) : new Vector(0, 1, 0);
        Vector u = dir.clone().crossProduct(up).normalize();
        Vector v = dir.clone().crossProduct(u).normalize();
        return new Vector[]{u, v};
    }

    /** A little ghost figure drawn from dust. */
    public static void silhouette(Location feet, Color color, float size) {
        World w = feet.getWorld();
        Particle.DustOptions d = dust(color, size);
        double yaw = Math.toRadians(feet.getYaw());
        double rx = Math.cos(yaw) * 0.38, rz = Math.sin(yaw) * 0.38;
        for (double y = 0; y <= 1.5; y += 0.25) w.spawnParticle(Particle.DUST, feet.clone().add(0, y, 0), 1, 0.03, 0, 0.03, 0, d);
        w.spawnParticle(Particle.DUST, feet.clone().add(rx, 1.2, rz), 1, 0, 0, 0, 0, d);
        w.spawnParticle(Particle.DUST, feet.clone().add(-rx, 1.2, -rz), 1, 0, 0, 0, 0, d);
        w.spawnParticle(Particle.DUST, feet.clone().add(rx * 0.5, 0.1, rz * 0.5), 1, 0, 0, 0, 0, d);
        w.spawnParticle(Particle.DUST, feet.clone().add(-rx * 0.5, 0.1, -rz * 0.5), 1, 0, 0, 0, 0, d);
        w.spawnParticle(Particle.END_ROD, feet.clone().add(0, 1.75, 0), 1, 0.05, 0.05, 0.05, 0);
    }

    public static void sound(Location l, Sound s, float vol, float pitch) { l.getWorld().playSound(l, s, vol, pitch); }
}

package com.astralis.kit;

import com.astralis.Astralis;
import com.astralis.core.Constellation;
import com.astralis.core.Damage;
import net.kyori.adventure.text.Component;
import net.kyori.adventure.text.format.NamedTextColor;
import org.bukkit.Location;
import org.bukkit.entity.Entity;
import org.bukkit.entity.LivingEntity;
import org.bukkit.entity.Player;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ThreadLocalRandom;

/** One constellation's passive, primary and ultimate. */
public abstract class Kit {
    protected final Astralis plugin;
    protected final Constellation constellation;

    protected Kit(Astralis plugin, Constellation constellation) {
        this.plugin = plugin;
        this.constellation = constellation;
    }

    public abstract double primaryBaseCooldown();

    /** Called every 0.5s while this sigil is the player's active one. */
    public void passive(Player p, double potency) { }

    /** @return true if the ability actually fired (so cooldown is consumed). */
    public abstract boolean primary(Player p, double potency);

    /** @return true if the ability actually fired (so Stardust is spent). */
    public abstract boolean ultimate(Player p, double potency);

    protected double v(String path, double def) {
        return plugin.val(constellation.name().toLowerCase() + "." + path, def);
    }

    /** Valid living targets whose body centre lies within radius of center. */
    protected List<LivingEntity> targets(Player caster, Location center, double radius) {
        List<LivingEntity> out = new ArrayList<>();
        double r2 = radius * radius;
        for (Entity e : center.getWorld().getNearbyEntities(center, radius, radius, radius)) {
            if (!Damage.valid(caster, e)) continue;
            LivingEntity le = (LivingEntity) e;
            if (le.getLocation().add(0, le.getHeight() / 2, 0).distanceSquared(center) <= r2) out.add(le);
        }
        return out;
    }

    protected static double rnd(double a, double b) { return ThreadLocalRandom.current().nextDouble(a, b); }

    protected void fail(Player p, String msg) { p.sendActionBar(Component.text(msg, NamedTextColor.RED)); }
}

/** Turns clicks into casts. Right-click = primary, sneak + right-click (or F) = Zenith. */
public final class AbilityListener implements Listener {
    private final Astralis plugin;

    public AbilityListener(Astralis plugin) { this.plugin = plugin; }

    @EventHandler(priority = EventPriority.HIGH)
    public void onInteract(PlayerInteractEvent e) {
        Action a = e.getAction();
        if (a == Action.PHYSICAL) return;
        ItemStack item = e.getItem();
        if (item == null) return;

        SigilManager sigils = plugin.sigils();
        boolean right = a == Action.RIGHT_CLICK_AIR || a == Action.RIGHT_CLICK_BLOCK;

        if (sigils.isCore(item)) {
            e.setCancelled(true);
            if (right && e.getHand() == EquipmentSlot.HAND) upgrade(e.getPlayer());
            return;
        }
        if (!sigils.isSigil(item)) return;

        Player p = e.getPlayer();
        // Let people still open chests, doors, etc. with a sigil in hand (unless sneaking).
        if (a == Action.RIGHT_CLICK_BLOCK && !p.isSneaking() && e.getClickedBlock() != null
                && e.getClickedBlock().getType().isInteractable()) return;

        e.setCancelled(true);
        if (!right || e.getHand() != EquipmentSlot.HAND) return;
        if (p.isSneaking()) plugin.abilities().ultimate(p);
        else plugin.abilities().primary(p);
    }

    /** F key (swap hands) is an alternate Zenith trigger - handy for Bedrock/Geyser players. */
    @EventHandler(priority = EventPriority.HIGH, ignoreCancelled = true)
    public void onSwap(PlayerSwapHandItemsEvent e) {
        if (!plugin.getConfig().getBoolean("general.ultimate-on-swap", true)) return;
        // getOffHandItem() is the item that is currently in the main hand
        if (!plugin.sigils().isSigil(e.getOffHandItem())) return;
        e.setCancelled(true);
        plugin.abilities().ultimate(e.getPlayer());
    }

    private void upgrade(Player p) {
        SigilManager sm = plugin.sigils();
        ItemStack off = p.getInventory().getItemInOffHand();
        Sigil s = sm.read(off);
        if (s == null) {
            p.sendActionBar(Component.text("Hold a Dim sigil in your off-hand.", NamedTextColor.RED));
            return;
        }
        if (s.tier() == Tier.RADIANT) {
            p.sendActionBar(Component.text("That sigil is already Radiant.", NamedTextColor.RED));
            return;
        }

        ItemStack main = p.getInventory().getItemInMainHand().clone();
        if (main.getAmount() > 1) { main.setAmount(main.getAmount() - 1); p.getInventory().setItemInMainHand(main); }
        else p.getInventory().setItemInMainHand(null);
        p.getInventory().setItemInOffHand(sm.upgrade(off));

        var c = s.constellation();
        p.showTitle(Title.title(
                c.text("✦✦ ASCENDED ✦✦"),
                Component.text(c.display() + " burns Radiant", NamedTextColor.GRAY),
                Title.Times.times(Duration.ofMillis(200), Duration.ofMillis(1800), Duration.ofMillis(500))));
        Fx.sound(p.getLocation(), Sound.BLOCK_BEACON_ACTIVATE, 1.3f, 1.4f);
        Fx.sound(p.getLocation(), Sound.ITEM_TOTEM_USE, 0.7f, 1.5f);

        final Particle.DustOptions dust = Fx.dust(c.bukkitColor(), 1.4f);
        new BukkitRunnable() {
            int t = 0;

            @Override
            public void run() {
                if (t >= 36 || !p.isOnline()) {

            }

/** Passive effects, the counter ring on melee/arrows, Stardust gain, and status handling. */
public final class CombatListener implements Listener {
    private static final Color TEAL = Color.fromRGB(0x6ff2c5);
    private final Astralis plugin;

    public CombatListener(Astralis plugin) { this.plugin = plugin; }

    private static boolean isMelee(EntityDamageEvent e) {
        return e.getCause() == EntityDamageEvent.DamageCause.ENTITY_ATTACK
                || e.getCause() == EntityDamageEvent.DamageCause.ENTITY_SWEEP_ATTACK;
    }

    private static Player attackerOf(org.bukkit.entity.Entity damager) {
        if (damager instanceof Player p) return p;
        if (damager instanceof Projectile pr && pr.getShooter() instanceof Player shooter) return shooter;
        return null;
    }

    @EventHandler(priority = EventPriority.HIGH, ignoreCancelled = true)
    public void onHit(EntityDamageByEntityEvent e) {
        if (Damage.isAbility() || !(e.getEntity() instanceof LivingEntity victim)) return;
        final Player attacker = attackerOf(e.getDamager());
        final boolean melee = isMelee(e);

        // ---------------- defensive side ----------------
        if (victim instanceof Player vp && !plugin.disabledWorld(vp.getWorld())) {
            Sigil vs = plugin.sigils().active(vp);
            double vpot = vs == null ? 1.0 : plugin.sigils().potency(vp, vs);

            // Gemini - Afterimage: chance to fully dodge a melee hit
            if (vs != null && vs.constellation() == Constellation.GEMINI && melee
                    && ThreadLocalRandom.current().nextDouble() < plugin.val("gemini.dodge-chance", 0.10) * vpot) {
                dodge(vp);
                e.setCancelled(true);
                return;
            }

            // Cassiopeia - Starlit Aegis: reflect a share of melee damage
            if (melee && plugin.statuses().has(vp.getUniqueId(), StatusManager.Status.AEGIS)
                    && e.getDamager() instanceof LivingEntity src && src != vp) {
                final double reflect = e.getDamage() * plugin.val("cassiopeia.aegis.reflect", 0.25) * vpot;
                Bukkit.getScheduler().runTask(plugin, () -> {
                    if (vp.isOnline() && src.isValid() && !src.isDead() && Damage.deal(vp, src, reflect)) {
                        Location l = src.getLocation().add(0, src.getHeight() / 2, 0);
                        l.getWorld().spawnParticle(Particle.ENCHANTED_HIT, l, 15, 0.3, 0.4, 0.3, 0.3);
                        Fx.sound(l, Sound.BLOCK_AMETHYST_BLOCK_CHIME, 1f, 1.6f);
                    }
                });
            }
        }

        // ---------------- offensive side ----------------
        if (attacker == null || attacker == victim || plugin.disabledWorld(attacker.getWorld())) return;

        Sigil as = plugin.sigils().active(attacker);
        double apot = as == null ? 1.0 : plugin.sigils().potency(attacker, as);
        double mult = Damage.modifier(attacker, victim);   // counter ring x Orion mark
        final double base = e.getDamage();

        if (as != null) {
            switch (as.constellation()) {
                case ORION -> {
                    // Steady Aim
                    if (e.getDamager() instanceof AbstractArrow)
                        mult *= 1 + plugin.val("orion.arrow-bonus", 0.15) * apot;
                }
                case ARIES -> {
                    // Momentum: sprinting melee hits hit harder and hurl enemies back
                    if (melee && attacker.isSprinting()) {
                        mult *= 1 + plugin.val("aries.momentum-bonus", 0.12) * apot;
                        Vector kb = victim.getLocation().toVector().subtract(attacker.getLocation().toVector()).setY(0);
                        if (kb.lengthSquared() > 0.01) {
                            final Vector push = kb.normalize().multiply(0.5).setY(0.22);
                            Bukkit.getScheduler().runTask(plugin, () -> {
                                if (victim.isValid()) victim.setVelocity(victim.getVelocity().add(push));
                            });
                        }
                    }
                }
                default -> { }
            }
        }

        // Gemini - Twin Echo: every melee hit repeats a moment later
        if (melee && plugin.statuses().has(attacker.getUniqueId(), StatusManager.Status.ECHO)) {
            final double echo = base * plugin.val("gemini.echo.ratio", 0.5) * apot;
            Bukkit.getScheduler().runTaskLater(plugin, () -> {
                if (attacker.isOnline() && victim.isValid() && !victim.isDead() && Damage.deal(attacker, victim, echo)) {
                    Location l = victim.getLocation().add(0, victim.getHeight() / 2, 0);
                    l.getWorld().spawnParticle(Particle.SWEEP_ATTACK, l, 1);
                    l.getWorld().spawnParticle(Particle.END_ROD, l, 8, 0.3, 0.3, 0.3, 0.1);
                    Fx.sound(l, Sound.ENTITY_PLAYER_ATTACK_SWEEP, 0.9f, 1.5f);
                }
            }, 6L);
        }

        e.setDamage(base * mult);
    }

    /** Stardust gain, using the damage actually dealt. */
    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onHitMonitor(EntityDamageByEntityEvent e) {
        if (Damage.isAbility() || !(e.getEntity() instanceof LivingEntity victim)) return;
        double dmg = e.getFinalDamage();
        if (dmg <= 0) return;
        Player attacker = attackerOf(e.getDamager());
        if (attacker != null && attacker != victim) gain(attacker, dmg, victim instanceof Player, true);
        if (victim instanceof Player vp) gain(vp, dmg, attacker != null, false);
    }

    private void gain(Player p, double dmg, boolean pvp, boolean dealt) {
        if (plugin.disabledWorld(p.getWorld()) || plugin.sigils().active(p) == null) return;
        var cfg = plugin.getConfig();
        double rate = cfg.getDouble(dealt ? "stardust.per-damage-dealt" : "stardust.per-damage-taken", dealt ? 0.8 : 0.5);
        double amt = Math.min(cfg.getDouble("stardust.per-hit-cap", 8), dmg * rate);
        if (!pvp) amt *= cfg.getDouble("stardust.mob-multiplier", 0.25);
        plugin.stardust().add(p, amt);
    }

    /** Phoenix - Undying Ember: a lethal hit is cheated (long cooldown). */
    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onLethal(EntityDamageEvent e) {
        if (!(e.getEntity() instanceof Player p) || plugin.disabledWorld(p.getWorld())) return;
        EntityDamageEvent.DamageCause cause = e.getCause();
        if (cause == EntityDamageEvent.DamageCause.VOID || cause == EntityDamageEvent.DamageCause.SUICIDE
                || cause == EntityDamageEvent.DamageCause.KILL) return;
        if (e.getFinalDamage() < p.getHealth()) return;
        Sigil s = plugin.sigils().active(p);
        if (s == null || s.constellation() != Constellation.PHOENIX || !plugin.enabled(Constellation.PHOENIX)) return;
        PhoenixKit kit = (PhoenixKit) plugin.abilities().kit(Constellation.PHOENIX);
        if (kit.tryRevive(p, plugin.sigils().potency(p, s), s.tier())) e.setCancelled(true);
    }

    /** Dragon dive / Rampart Charge grant brief fall immunity. */
    @EventHandler(ignoreCancelled = true)
    public void onFall(EntityDamageEvent e) {
        if (e.getCause() == EntityDamageEvent.DamageCause.FALL
                && plugin.statuses().has(e.getEntity().getUniqueId(), StatusManager.Status.NO_FALL)) {
            e.setCancelled(true);
        }
    }

    private void dodge(Player vp) {
        Location l = vp.getLocation();
        Fx.silhouette(l, TEAL, 1f);
        vp.getWorld().spawnParticle(Particle.END_ROD, l.clone().add(0, 1, 0), 15, 0.3, 0.6, 0.3, 0.08);
        Fx.sound(l, Sound.ENTITY_ILLUSIONER_MIRROR_MOVE, 1f, 1.8f);
        vp.addPotionEffect(new PotionEffect(PotionEffectType.SPEED, 40, 1, false, true));
    }

    // ------------------------------------------------------------------ join / quit
    @EventHandler
    public void onJoin(PlayerJoinEvent e) {
        Player p = e.getPlayer();
        plugin.stardust().load(p);
        plugin.trust().load(p);
        plugin.sigils().discover(p);
        plugin.events().onJoin(p);

        if (!plugin.getConfig().getBoolean("general.starter-sigil", true)) return;
        if (p.getPersistentDataContainer().has(plugin.keys().received, PersistentDataType.BYTE)) return;
        Bukkit.getScheduler().runTaskLater(plugin, () -> {
            if (p.isOnline() && !p.getPersistentDataContainer().has(plugin.keys().received, PersistentDataType.BYTE)) {
                plugin.sigils().awaken(p, plugin.sigils().pickBalanced());
            }
        }, 60L);
    }

    @EventHandler
    public void onQuit(PlayerQuitEvent e) {
        Player p = e.getPlayer();
        plugin.stardust().save(p);
        plugin.trust().save(p);
        plugin.statuses().clearAll(p.getUniqueId());
        lastWarn.remove(p.getUniqueId());
    }

    @EventHandler

        Player killer = victim.getKiller();

        var cfg = plugin.getConfig();
        boolean dropPvp = cfg.getBoolean("general.drop-on-pvp-death", true);
        boolean keepNatural = cfg.getBoolean("general.keep-on-natural-death", true);
        SigilManager sm = plugin.sigils();

        if (!e.getKeepInventory()) {
            List<ItemStack> found = new ArrayList<>();
            for (ItemStack it : e.getDrops()) if (sm.isSigil(it)) found.add(it);
            if (!found.isEmpty()) {
                if (pvp && dropPvp) {
                    e.getDrops().removeIf(sm::isSigil);
                    for (ItemStack it : found) dropSigil(victim, killer, it);
                } else if (keepNatural) {
                    e.getDrops().removeIf(sm::isSigil);
                    e.getItemsToKeep().addAll(found);
                }
            }
        } else if (pvp && dropPvp) {
            PlayerInventory inv = victim.getInventory();
            for (int i = 0; i < inv.getSize(); i++) {
                ItemStack it = inv.getItem(i);
                if (sm.isSigil(it)) {
                    inv.setItem(i, null);
                    dropSigil(victim, killer, it);
                }
            }
        }

        // Stardust: the killer gains, the fallen keeps only a share
        if (pvp && sm.active(killer) != null && !plugin.disabledWorld(killer.getWorld())) {
            plugin.stardust().add(killer, cfg.getDouble("stardust.kill-bonus", 25));
        }
        plugin.stardust().set(victim, plugin.stardust().get(victim) * cfg.getDouble("stardust.death-retention", 0.5));
        plugin.statuses().clearAll(victim.getUniqueId());
    }

    private void dropSigil(Player victim, Player killer, ItemStack stack) {
        Item item = victim.getWorld().dropItemNaturally(victim.getLocation(), stack);
        plugin.sigils().protectDrop(item);
        Sigil s = plugin.sigils().read(stack);
        if (s != null && plugin.getConfig().getBoolean("general.broadcast-seizure", true)) {
            Constellation c = s.constellation();
            Bukkit.broadcast(Astralis.mm("<gold>✦ <white>" + Astralis.MM.escapeTags(victim.getName()) + "<gray>'s "
                    + c.tag() + "<bold>" + c.display() + "</bold><gray> sigil was torn loose by <white>"
                    + Astralis.MM.escapeTags(killer.getName()) + "<gray>! It burns where it fell."));
        }
    }

    @EventHandler
    public void onDrop(PlayerDropItemEvent e) {
        if (plugin.sigils().isSigil(e.getItemDrop().getItemStack())) plugin.sigils().protectDrop(e.getItemDrop());
    }

    @EventHandler
    public void onEntitiesLoad(EntitiesLoadEvent e) {
        for (Entity en : e.getEntities()) {
            if (en instanceof Item item && plugin.sigils().isSigil(item.getItemStack())) plugin.sigils().protectDrop(item);
        }
    }

    // ------------------------------------------------------------------ pickup rules
    @EventHandler(priority = EventPriority.HIGH, ignoreCancelled = true)
    public void onPickup(EntityPickupItemEvent e) {
        ItemStack st = e.getItem().getItemStack();

        if (plugin.events().isShard(st)) {
            e.setCancelled(true);
            if (e.getEntity() instanceof Player p) plugin.events().collect(p, e.getItem());
            return;
        }

        SigilManager sm = plugin.sigils();
        Sigil s = sm.read(st);
        if (s == null) return;
        if (!(e.getEntity() instanceof Player p)) { e.setCancelled(true); return; }

        int max = plugin.getConfig().getInt("general.max-sigils-carried", 2);
        if (sm.carried(p) >= max) {
            e.setCancelled(true);
            long now = System.currentTimeMillis();
            if (now - lastWarn.getOrDefault(p.getUniqueId(), 0L) > 2000) {
                lastWarn.put(p.getUniqueId(), now);
                p.sendActionBar(Component.text("You can only carry " + max + " sigils.", NamedTextColor.RED));
            }
            return;
        }
        sm.tracked().remove(e.getItem().getUniqueId());

        boolean own = s.owner() == null || s.owner().equals(p.getUniqueId());
        Constellation c = s.constellation();
        if (!own) {
            String ownerName = sm.ownerName(st);
            p.sendActionBar(Astralis.mm(c.tag() + "✦ <gray>This sigil is <red>unattuned<gray> to you: reduced power."));
            if (plugin.getConfig().getBoolean("general.broadcast-seizure", true)) {
                Bukkit.broadcast(Astralis.mm("<gold>✦ <white>" + Astralis.MM.escapeTags(p.getName()) + "<gray> seized the "
                        + c.tag() + "<bold>" + c.display() + "</bold><gray> sigil of <white>"
                        + Astralis.MM.escapeTags(ownerName == null ? "an unknown soul" : ownerName) + "<gray>."));
            }
        } else {
            p.sendActionBar(Astralis.mm(c.tag() + "✦ <gray>Your sigil returns to you."));
        }
    }

    // ------------------------------------------------------------------ storage protection
    private boolean storageBlocked() { return !plugin.getConfig().getBoolean("general.allow-storage", false); }

    private static boolean playerOnly(Inventory top) {
        InventoryType t = top.getType();
        return t == InventoryType.CRAFTING || t == InventoryType.CREATIVE;
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onClick(InventoryClickEvent e) {
        if (!storageBlocked()) return;
        Inventory top = e.getView().getTopInventory();
        if (playerOnly(top)) return;
        boolean inTop = e.getRawSlot() >= 0 && e.getRawSlot() < top.getSize();
        SigilManager sm = plugin.sigils();

        if (inTop && sm.isSigil(e.getCursor())) { e.setCancelled(true); return; }
        if (!inTop && e.isShiftClick() && sm.isSigil(e.getCurrentItem())) { e.setCancelled(true); return; }
        if (inTop && e.getClick() == ClickType.NUMBER_KEY) {
            ItemStack hot = e.getWhoClicked().getInventory().getItem(e.getHotbarButton());
            if (sm.isSigil(hot)) { e.setCancelled(true); return; }
        }
        if (inTop && e.getClick() == ClickType.SWAP_OFFHAND && sm.isSigil(e.getWhoClicked().getInventory().getItemInOffHand())) {
            e.setCancelled(true);
        }
    }

    @EventHandler(priority = EventPriority.HIGHEST, ignoreCancelled = true)
    public void onDrag(InventoryDragEvent e) {
        if (!storageBlocked()) return;
        Inventory top = e.getView().getTopInventory();
        if (playerOnly(top) || !plugin.sigils().isSigil(e.getOldCursor())) return;
        for (int slot : e.getRawSlots()) {
            if (slot < top.getSize()) { e.setCancelled(true); return; }
        }
    }

    @EventHandler(ignoreCancelled = true)
    public void onHopper(InventoryMoveItemEvent e) {
        if (storageBlocked() && plugin.sigils().isSigil(e.getItem())) e.setCancelled(true);
    }

    @EventHandler(ignoreCancelled = true)
    public void onHopperPickup(InventoryPickupItemEvent e) {
        if (plugin.sigils().isSigil(e.getItem().getItemStack())) e.setCancelled(true);
    }

    // ------------------------------------------------------------------ crafting / anvil / frames
    @EventHandler
    public void onPrepareCraft(PrepareItemCraftEvent e) {
        Recipe r = e.getRecipe();
        if (r instanceof Keyed k && k.getKey().equals(plugin.keys().recipeCore)) return;
        for (ItemStack it : e.getInventory().getMatrix()) {
            if (it != null && (plugin.sigils().isSigil(it) || plugin.sigils().isCore(it) || plugin.events().isShard(it))) {
                e.getInventory().setResult(null);
                return;
            }
        }
    }

    @EventHandler
    public void onPrepareAnvil(PrepareAnvilEvent e) {
        for (int i = 0; i < 2; i++) {
            ItemStack it = e.getInventory().getItem(i);
            if (it != null && (plugin.sigils().isSigil(it) || plugin.sigils().isCore(it))) {
                e.setResult(null);
                return;
            }
        }
    }

    @EventHandler(ignoreCancelled = true)
    public void onInteractEntity(PlayerInteractEntityEvent e) {
        Entity t = e.getRightClicked();
        if (!(t instanceof ItemFrame || t instanceof ArmorStand || t instanceof Allay)) return;
        ItemStack hand = e.getPlayer().getInventory().getItem(e.getHand());
        if (plugin.sigils().isSigil(hand) || plugin.sigils().isCore(hand)) e.setCancelled(true);
    }
}

package com.astralis.task;

import com.astralis.Astralis;
import com.astralis.core.Constellation;
import com.astralis.core.Sigil;
import com.astralis.core.Tier;
import org.bukkit.Bukkit;
import org.bukkit.GameMode;
import org.bukkit.Location;
import org.bukkit.Particle;
import org.bukkit.entity.Entity;
import org.bukkit.entity.Item;
import org.bukkit.entity.Player;
import org.bukkit.potion.PotionEffectType;
import org.bukkit.scheduler.BukkitRunnable;

import java.util.Iterator;
import java.util.UUID;

/** Every 2 ticks: ambient sparkle. Every 10: passives, HUD, dropped-sigil beacons. */
public final class StarTask extends BukkitRunnable {
    private final Astralis plugin;
    private int tick = 0;

    public StarTask(Astralis plugin) { this.plugin = plugin; }

    @Override
    public void run() {
        tick += 2;
        boolean slow = tick % 10 == 0;
        boolean ambient = plugin.getConfig().getBoolean("general.ambient-particles", true);

        for (Player p : Bukkit.getOnlinePlayers()) {
            if (p.isDead() || p.getGameMode() == GameMode.SPECTATOR || plugin.disabledWorld(p.getWorld())) continue;
            Sigil s = plugin.sigils().active(p);
            if (s == null) continue;
            if (ambient && !p.isInvisible() && !p.hasPotionEffect(PotionEffectType.INVISIBILITY)) ambient(p, s);
            if (slow) {
                plugin.abilities().kit(s.constellation()).passive(p, plugin.sigils().potency(p, s));
                hud(p);
            }
        }
        if (slow) beacons();
        if (tick % 100 == 0) plugin.statuses().purge();
    }

    private void ambient(Player p, Sigil s) {
        double a = tick * 0.22;
        Location base = p.getLocation().add(0, 1.0, 0);
        var dust = new Particle.DustOptions(s.constellation().bukkitColor(), 0.8f);
        for (int i = 0; i < 2; i++) {
            double ang = a + i * Math.PI;
            Location l = base.clone().add(Math.cos(ang) * 0.75, Math.sin(a * 0.6 + i) * 0.5, Math.sin(ang) * 0.75);
            p.getWorld().spawnParticle(Particle.DUST, l, 1, 0, 0, 0, 0, dust);
        }
        if (s.tier() == Tier.RADIANT && tick % 6 == 0) {
            p.getWorld().spawnParticle(Particle.END_ROD, base.clone().add(0, 0.9, 0), 1, 0.25, 0.1, 0.25, 0.01);
        }
    }

    private void hud(Player p) {
        Sigil hs = plugin.sigils().read(p.getInventory().getItemInMainHand());
        if (hs == null) return;
        Constellation c = hs.constellation();
        UUID id = p.getUniqueId();
        double energy = plugin.stardust().get(p);
        double max = plugin.stardust().max();
        int cost = plugin.ultCost(hs.tier());
        int filled = (int) Math.round(10 * Math.min(1.0, energy / max));

        double pcd = plugin.cooldowns().remaining(id, c.name() + ":P");
        double ucd = plugin.cooldowns().remaining(id, c.name() + ":U");
        String primary = pcd > 0 ? "<gray>◇ " + (int) Math.ceil(pcd) + "s" : "<green>◆ Ready";
        String ult;
        if (ucd > 0) ult = "<gray>★ " + (int) Math.ceil(ucd) + "s";
        else if (energy >= cost) ult = "<gold><bold>★ ZENITH READY";
        else ult = "<gray>★ " + (int) energy + "/" + cost;

        String text = c.tag() + "✦ " + c.display() + " <dark_gray>▏ " + c.tag() + "▰".repeat(filled)
                + "<dark_gray>" + "▱".repeat(10 - filled) + " <dark_gray>▏ " + primary + " <dark_gray>▏ " + ult;
        p.sendActionBar(Astralis.mm(text));
    }

    /** A light pillar over every sigil lying on the ground. */
    private void beacons() {
        Iterator<UUID> it = plugin.sigils().tracked().iterator();
        while (it.hasNext()) {
            Entity e = Bukkit.getEntity(it.next());
            if (!(e instanceof Item item) || !item.isValid()) { it.remove(); continue; }
            Sigil s = plugin.sigils().read(item.getItemStack());
            if (s == null) { it.remove(); continue; }
            Location l = item.getLocation();
            var dust = new Particle.DustOptions(s.constellation().bukkitColor(), 1.3f);
            for (int i = 0; i < 14; i++) {
                Location at = l.clone().add(0, 0.3 + i * 0.5, 0);
                l.getWorld().spawnParticle(Particle.END_ROD, at, 1, 0.02, 0, 0.02, 0);
                if (i % 2 == 0) l.getWorld().spawnParticle(Particle.DUST, at, 1, 0.15, 0, 0.15, 0, dust);
            }
        }
    }
}
